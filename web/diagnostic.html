// Quick Bottie Status Check & Enable
async function quickBottieCheck() {
    try {
        // Test multiple credential combinations
        const credentials = [
            { username: 'admin', password: 'AdminPass456!' },
            { username: 'superuser', password: 'SuperSecret123!' },
            { username: 'admin', password: 'admin' },
            { username: 'superuser', password: 'superuser' }
        ];
        
        let authToken = null;
        
        for (const cred of credentials) {
            try {
                const response = await fetch('/auth/login', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(cred)
                });
                
                if (response.ok) {
                    const data = await response.json();
                    authToken = data.access_token;
                    console.log('✅ Login successful with:', cred.username);
                    break;
                }
            } catch (e) {
                console.log('❌ Failed:', cred.username);
            }
        }
        
        // Check current status
        const statusResponse = await fetch('/status');
        const status = await statusResponse.json();
        console.log('📊 Current Status:', status);
        
        // Try to enable execution if we have auth
        if (authToken) {
            const enableResponse = await fetch('/exec/enable', {
                method: 'POST',
                headers: { 'Authorization': `Bearer ${authToken}` }
            });
            
            if (enableResponse.ok) {
                console.log('⚡ Execution enabled successfully!');
            } else {
                console.log('⚠️ Enable execution failed:', enableResponse.status);
            }
        } else {
            console.log('⚠️ Could not authenticate - execution may already be enabled');
        }
        
        return status;
    } catch (error) {
        console.error('❌ Error:', error);
    }
}

// Run the check
quickBottieCheck();
